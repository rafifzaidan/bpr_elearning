% Penyetelan agar listings memotong baris secara otomatis jika terlalu panjang
\lstset{
    basicstyle=\ttfamily\small,
    breaklines=true,
    breakatwhitespace=false,
    breakautoindent=true,
    frame=single,
    showstringspaces=false,
    keywordstyle=\color{blue},
    commentstyle=\color{gray},
    stringstyle=\color{purple}
}

% 3. Paket Enumitem untuk Kustomisasi Tingkat List/Poin
\usepackage{enumitem}

\setlist{itemsep=2pt, parsep=0pt, topsep=4pt, partopsep=0pt}

% 4. Penomoran Bab, Section, dan Subsection
\setcounter{secnumdepth}{3} % Penomoran tingkat bab hingga subsection/subsubsection
\setcounter{tocdepth}{2}    % Daftar isi menampilkan hingga subsection

% 5. Header & Footer Halaman (Twoside Page Numbers)
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{} % Bersihkan semua header/footer default
\fancyfoot[LE,RO]{\thepage} % Nomor halaman di Kiri untuk halaman Genap (Even), Kanan untuk halaman Ganjil (Odd)
\renewcommand{\headrulewidth}{0pt} % Hapus garis atas header
\renewcommand{\footrulewidth}{0pt} % Hapus garis bawah footer

% Menyetel halaman pembuka bab agar tetap menggunakan gaya penomoran yang konsisten
\fancypagestyle{plain}{
    \fancyhf{}
    \fancyfoot[LE,RO]{\thepage}
    \renewcommand{\headrulewidth}{0pt}
}

% 6. Penanganan Halaman Kosong pada Twoside
\makeatletter
\def\cleardoublepage{\clearpage\if@twoside \ifodd\c@page\else

  \end{center}
  \vspace{\fill}
  \thispagestyle{empty}
  \newpage
  \if@twocolumn\hbox{}\newpage\fi\fi\fi}
\makeatother
